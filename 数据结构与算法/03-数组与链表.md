 方法 2：除自身以外数组的乘积可以看成两部分：元素自身左边、右边其他所有元素的乘积，而最终结果是这两部分的乘积。

 ```C++
class Solution{
public:
    vector<int> productExceptSelf(vector<int> &nums){
        vector<int> right(nums.size(), 0), left(nums.size(), 0);
        vector<int> res(nums.size(), 0);
        int product = 0;

        left[0] = 1;
        product = 1;
        for(int i = 0;i < nums.size(); i++){
            product *= nums[i];
            if(i + 1 < nums.size()) left[i + 1] = product;
        }

        right[nums.size() - 1] = 1;
        product = 1;
        for(int i = nums.size() - 1; i >= 0; i--){
            product *= nums[i];
            if(i - 1 >= 0) right[i - 1] = product;
        }

        for(int i = 0; i < nums.size(); i++){
            res[i] = left[i] * right[i];
        }
        
        return res;
    }
 }
 ```

方法 2：在方法 1 中需要额外申请 `O(m + n)` 的空间来记录行和列的置零情况，可以直接将记录内容放到矩阵
的第 0 行和第 0 列来实现只需要常数空间来存储若干变量。由于第 0 行和第 0 列的相关数据可能会被覆盖掉，
因此需要首先将第 0 行和第 0 列的置零情况记录下来。总的来说就是：从第 1 行和第 1 列的位置起，`i` 行/列
的置零情况被记录在 `matrix[][i] or matrix[0][i]` 位置，可以这样记录的原因是：某位置是否要被置零存在
两种情况，一是该位置所在行某元素为零，二是该位置所在列某元素为零。因此对于第 0 行或第 0 列的元素，在对
应的行或列中存在 0 元素时将会被置零，同时也相当于记录了这一行应该被置零；反之，该元素保持原有值。因此，
只有需要被置零的元素被操作了，其余保持原来的。
```C++
class Solution{
public:
    void setZeros(vector<vector<int>> &matrix){
        int row = matrix.size();
        int col = matrix[0].size();
        bool flagRow0 = false;
        bool flagCol0 = false;

        for(int i = 0; i < row; i++){
            if(!matrix[i][0]) flagCol0 = true;
        }
        for(int j = 0; j < col; j++){
            if(!matrix[0][j]) flagRow0 = true;
        }

        // 这里行和列都需要从 1 开始
        for(int i = 1; i < row; i++){
            for(int j = 1; j < col; j++){
                if(!matrix[i][j]) matrix[0][j] = matrix[i][0] = 0;
            }
        }

        // 这里行和列都需要从 1 开始
        for(int i = 1; i < row; i++){
            for(int j = 1; j < col; j++){
                if(!matrix[0][j] || !matrix[i][0]) matrix[i][j] = 0;
            }
        }

        if(flagRow0){
            for(int j = 0; j < col; j++){
                matrix[0][j] = 0;
            }
        }
        if(flagCol0){
            for(int i = 0; i < row; i++){
                matrix[i][0] = 0;
            }
        }
    }
}
```

```C++
/* 搜索旋转排序数组 
 * 旋转后的数组通过中点位置可以将数组分为有序区间和无序区间两部分
 * 通过中点与端点的大小关系可以判断中点至端点（或端点至中点）这段区间是否为有序的
 * 再判断 target 值是否在有序区间内来缩小搜索范围
 * 在：     缩小范围至有序区间（变成二分查找）
 * 不在：   缩小范围至无序区间（相当于缩小区间的进一步循环）
 * 循环上述步骤，直到搜索结束（结束条件与二分查找一致）。
 */
 class Solution{
 public:
    int search(vector<int>& nums, int target){
        int len = nums.size();
        if(len == 1) return nums[0] == target ? 0 : -1;

        int left = 0, right = len - 1;
        while(left <= right){
            int mid = left + (right - left) / 2;
            if(nums[mid] == target) return mid;
            if(nums[left] <= nums[mid]){
                if(nums[left] <= target && target <= nums[mid]){
                    right = mid - 1;
                }else{
                    left = mid + 1;
                }
            }else{
                if(nums[mid] <= target && target <= nums[right]){
                    left = mid + 1;
                }else{
                    right = mid - 1;
                }
            }
        }
        return -1;
    }
 }
```
12. 寻找两个正序数组的中位数
方法 1：[Leetcode 题解的评论思路](https://leetcode.cn/problems/median-of-two-sorted-arrays/solutions/258842/xun-zhao-liang-ge-you-xu-shu-zu-de-zhong-wei-s-114/?envType=study-plan-v2&envId=top-100-liked)："两个人拿着不同长度有弹性的绳子，绳子上从左往右按小到大的顺序写了数字；两人拉开了中间位置的数字比大小；输了的往右移，赢了的往左移（输了的右边的长度和赢了的左边长度，哪个小移小的一半）(你输了你拿个大的数出来比比，我拿个小的跟你比）保证两个人左手边的长度和等于两个绳子的总长度的一半；"：没太明白如何确定某个位置是否为中位数位置，因为不能通过在两个数组中各取一部分，然后让这两部分长度之和等于总长度的一半来判断当前位置是否为中位数位置；
方法 2：[leetcode 题解一，二分查找](https://leetcode.cn/problems/median-of-two-sorted-arrays/solutions/258842/xun-zhao-liang-ge-you-xu-shu-zu-de-zhong-wei-s-114/?envType=study-plan-v2&envId=top-100-liked)：从两有序数组长度推导出中位数的位置，再将其二等分到对于的数组中，并对相应的位置上的元素进行比较，舍去两者中较小值及其右边部分区间，于是总的中位数区间便缩小了一半。需要特殊处理的是：1. 中位数二等分后对应到数组中的位置不能越过边界；2. 由于 1 的存在，实际舍去的数量要分情况而定，不能直接更新为当前中位数区间值的一般；3. 当两数组中其中一个达到边界时即可结束循环； 4. 当中位数区间值为 1 时也可结束循环，此时结果为两数组当前索引位置对应值中较小的那一个；
```C++
class Solution {
public:
    int getKthElement(const vector<int>& nums1, const vector<int>& nums2, int k) {
        // 寻找由排序数组 nums1 和 nums2 组成的序列中第 k 个最小值的位置
        // k 的有效值从 1 开始
        // 通过不断缩小需要搜索的区间来找到 kth
        // 缩小的方式是：正常情况下，每次搜索都可以将区间范围，即 k 的值缩小到一半
        int len1 = nums1.size();
        int len2 = nums2.size();

        // 通过 index1 和 index 2 来记录 nums1 和 nums2 中的标记位置
        // 以 index1 和 index2 为左端点，以当前区间长度一半为偏移，该位置及其左边部分是被考虑排除的区间，即缩小的 k 的区间
        // index1 和 index2 从索引 0 位置开始
        int index1 = 0;
        int index2 = 0;
        int curk = k;

        while(true){
            // 结束条件：
            // 1. 完全排除 nums1
            // 2. 完全排序 nums2
            // 3. 搜索结束，即 curk = 1
            if(index1 == len1)
                return nums2[k - len1 - 1];
            if(index2 == len2)
                return nums1[k - len2 - 1];
            if(curk == 1)
                return min(nums1[index1], nums2[index2]);
            
            
            int nextIndex1 = min(index1 + curk / 2 - 1, len1 - 1);
            int nextIndex2 = min(index2 + curk / 2 - 1, len2 - 1);
            int value1 = nums1[nextIndex1];
            int value2 = nums2[nextIndex2];
            if(value1 < value2){
                // nextIndex1 及其左边部分可以排除
                curk -= (nextIndex1 - index1 + 1);
                index1 = nextIndex1 + 1;
            }else{
                // index2 及其左边部分可以排除
                curk -= (nextIndex2 - index2 + 1);
                index2 = nextIndex2 + 1;                
            }
        }
    }

    double findMedianSortedArrays(vector<int>& nums1, vector<int>& nums2) {
        int len1 = nums1.size(), len2 = nums2.size();
        int odd = (len1 + len2) % 2;
        int kth = (len1 + len2 + 1) / 2;
        if(odd){
            return getKthElement(nums1, nums2, kth);
        }else{
            return (getKthElement(nums1, nums2, kth) + getKthElement(nums1, nums2, kth + 1)) / (double)2;
        }
    }
};
```

```C++
class Solution{
public:
    Node* copyRandomList(Node *head){
        if(!head) return nullptr;
        Node *cur = new Node(0);
        Node *newHead = cur;

        while(head){
            cur = head->val;
            if(head->next) cur->next = new Node(0);
            if(head->random) cur->random = new Node(head->random->val);

            head = head->next;
        }

        return newHead;
    }
};

/* 方法 2：通过哈希表先复制原始链表中的每个节点（此时并没有将新建节点连接在一起），再次从开始遍历原始链表，从哈希表中取出复制的节点并连接和指定 random，此时 random 所指对象存在并且在链表中。 */
class Solution{
public:
    Node* copyRandomList(Node *head){
        if(!head) return nullptr;
        unordered_map<Node*, Node*> mp;
        Node *oldHead = head;

        // 复制节点
        while(head){
            // mp.insert({head, new Node(head->val)});
            mp[head] = new Node(head->val);
            head = head->next;
        }

        // 建立新节点间的联系
        head = oldHead;
        while(head){
            mp[head]->next = mp[head->next];
            mp[head]->random = mp[head->random];
            head = head->next;
        }

        return mp[oldHead];
    }
};

/* 方法 3：通过先将复制的节点链接在原始节点之后创一个新链表，在这个链表中所有复制节点都处于奇数位置（以 0 开始），可以通过复制节点的前一节点的随机节点情况很方便地确定复制节点的随机节点。 当原始节点的随机节点为空时，复制节点的随机节点也为空；当原始节点的随机节点不为空时，复制节点的随机节点指向原始随机节点的下一节点。*/
class Solution{
public:
    Node* copyRandomList(Node *head){
        // 1. 先将原始链表的节点的复制节点链接在该节点的后面
        if(!head) return nullptr;
        for(Node *cur = head; cur != nullptr; cur = cur->next->next){
            Node *newNode = new Node(cur->val);
            newNode->next = cur->next;
            cur->next = newNode;
        }
        // 2. 更新复制节点的 random 节点，random 节点指向当前复制节点的原始节点（即前一节点）的 random 之后
        for(Node *cur = head; cur != nullptr; cur = cur->next->next){
            cur->next->random = cur->random == nullptr ? nullptr : cur->random->next;
        }
        // 3. 从原始和复制节点组成的链表中拆分出来，得到原始链表和复制链表
        Node *newHead = head->next;
        for(Node *cur = head; cur != nullptr; cur = cur->next){
            Node *copyNode = cur->next;
            cur->next = cur->next->next;
            copyNode->next = cur->next == nullptr ? nullptr : cur->next->next;
        }

        return newHead;
    }
};
```

8. 题中进一步的要求：在 `O(nlogn)` 时间复杂度和常数级空间复杂度下，对链表进行排序。1. 对于时间复杂度为 `O(logn)` 可以联想到二分查找和递归等方法；2. 排序链表问题可以拆分为：排序子链表 + 合并有序的链表；3. 将原链表拆分为子链表再进行排序的方式——归并排序。根据子链表的拆分方式可以分为：自顶向下的归并排序（将原链表拆分成两部分，递归调用排序链表再合并，深度递归）和自底向上的归并排序（先将原链表中的元素两两一组进行排序，从而保证后序子链表合并时是有序链表。增加子链表的长度再次遍历，重复上述操作直到结束；横向遍历）。在深度递归中，需要使用 `O(logn)`额外（栈）空间；在横向遍历中，使用 `O(1)` 的空间复杂度。4. **插入排序的时间复杂度是 `O(n^2)`**，其中 `n` 是链表的长度。时间复杂度是 `O(nlogn)` 的排序算法包括归并排序、堆排序、快速排序（快速排序的最差时间复杂度是 `O(n^2)`），**其中最适合链表的排序算法是归并排序**。
```C++
/* 方法 1：自顶向下归并排序 
 * 时间复杂度为 `O(nlogn)`；空间复杂度为 `O(logn)`
 */
class Solution{
private:
    /* 找链表的中点 
     * 链表长度为奇数时，中点位于中心
     * 链表长度为偶数时，中点位于中心偏左位置
     */
    ListNode* findMidNode(ListNode *head){
        // 通过快慢指针寻找中点
        // 当 fast 的起始位置为 head 时，中点位置是中心或者中心偏右
        // 当 fast 的起始位置为 head->next 时，中点位置是中心或者中心偏左
        if(!head) return nullptr;
        ListNode *slow = head, *fast = head->next;
        while(fast && fast->next){
            fast = fast->next->next;
            slow = slow->next;
        }
        return slow;
    }

    /* 合并有序链表 */
    ListNode* mergeOrderedList(ListNode *head1, ListNode *head2){
        ListNode *dummy = new ListNode(0);
        ListNode *cur = dummy;

        while(head1 && head2){
            if(head1->val < head2->val){
                cur->next = head1;
                head1 = head1->next;
            }else{
                cur->next = head2;
                head2 = head2->next;
            }
            cur = cur->next;
        }

        if(head1) cur->next = head1;
        if(head2) cur->next = head2;

        return dummy->next;
    }
public:
    /* 深度递归进行归并排序 */
    ListNode* sortList(ListNode *head){
        // 当前链表只有一个元素时直接返回，nullptr 也算作一个元素
        if(!head || !head->next) return head;

        ListNode *mid = findMidNode(head);
        ListNode *nextHead = mid->next;
        // 断开链表
        mid->next = nullptr;

        // 递归调用，排序左右部分
        ListNode *left = sortList(head);
        ListNode *right = sortList(nextHead);

        // 合并左右部分
        return mergeOrderedList(left, right);
    }
};
/* 方法 2：自底向上归并排序 
 * 时间复杂度：O(nlogn)；空间复杂度：O(1)
 */
class Solution{
private:
    ListNode* mergeOrderedList(ListNode *head1, ListNode *head2){
        ListNode *dummy = new ListNode(0);
        ListNode *cur = dummy;

        while(head1 && head2){
            if(head1->val < head2->val){
                cur->next = head1;
                head1 = head1->next;
            }else{
                cur->next = head2;
                head2 = head2->next;
            }
            cur = cur->next;
        }
        if(head1) cur->next = head1;
        if(head2) cur->next = head2;
        
        return dummy->next;
    }
public:
    ListNode *sortList(ListNode *head){
        if(!head) return nullptr;

        
        // 获取链表长度
        ListNode *cur = head;
        int len = 0;
        while(cur){
            len++;
            cur = cur->next;
        }

        ListNode *dummy = new ListNode(0);
        dummy->next = head;
        // 先将原链表中节点一一合并成有序链表再两两合并...直到结束
        for(int subLen = 1; subLen < len; subLen <<= 1){
            ListNode *pre = dummy; 
            cur = dummy->next;
            while(cur != nullptr){
                ListNode *head1 = cur;
                for(int i = 1; i < subLen && cur->next != nullptr; i++){
                    cur = cur->next;
                }

                ListNode *head2 = cur->next;
                cur->next = nullptr;
                cur = head2;
                for(int i = 1; i < subLen && cur != nullptr && cur->next != nullptr; i++){
                    cur = cur->next;
                }

                ListNode *next = nullptr;
                if(cur != nullptr){
                    next = cur->next;
                    cur->next = nullptr;
                }

                ListNode *merged = mergeOrderedList(head1, head2);
                pre->next = merged;
                while(pre->next != nullptr){
                    pre = pre->next;
                }
                cur = next;
            }
        }

        return dummy->next;
    }
};
```

10. `LRU, Least Recent Using，最近最少使用缓存`。**`LRU` 缓存机制可以通过哈希表辅以双向链表实现**，用一个哈希表和一个双向链表维护所有在缓存中的键值对。其中：1. 双向链表按照被使用的顺序存储了这些键值对，靠近头部的键值对是最近使用的，而靠近尾部的键值对是最久未使用的；2. 哈希表，通过缓存数据的键映射到其在双向链表中的位置。于是，**可以使用哈希表进行定位，找出缓存项在双向链表中的位置**，随后将其移动双向链表的头部。**由于是双向链表，因此可以在 `O(1)` 时间内完成 `get` 和 `put` 操作**。**在双向链表的实现中，使用一个伪头部和伪尾部标记界限，这样在添加节点和删除节点的时候就不需要检查相邻节点是否存在**
```C++
struct DLinkedNode{
    int key, value;
    DLinkedNode *prev;
    DLinkedNode *next;
    DLinkedNode() : key(0), value(0), prev(nullptr), next(nullptr){}
    DLinkedNode(int _key, int _value) : key(_key), value(_value), prev(nullptr), next(nullptr){}
};

class LRUCache{
private:
    DLinkedNode *_head;
    DLinkedNode *_tail;
    int _size;
    int _capacity;
    unordered_map<int, DLinkedNode*> _cache;

    void moveToHead(DLinkedNode *node){
        removeNode(node);
        addToHead(node);
    }

    void removeNode(DLinkedNode *node){
        node->prev->next = node->next;
        node->next->prev = node->prev;
    }

    void addToHead(DLinkedNode *node){
        node->prev = _head;
        node->next = _head->next;
        _head->next->prev = node;
        _head->next = node;
    }

    DLinkedNode* removeTail(){
        DLinkedNode *node = _tail->prev;
        removeNode(node);
        return node;
    }

public:
    LRUCache(int capacity){
        _capacity   = capacity;
        _size       = 0;
        _head        = new DLinkedNode();
        _tail        = new DLinkedNode();
        _head->next  = _tail;
        _tail->prev  = _head;
    }

    int get(int key){
        if(!_cache.count(key)) return -1;
        DLinkedNode *node = _cache[key];
        moveToHead(node);
        return node->value;
    }

    void put(int key, int value){
        if(!_cache.count(key)){
            // key 不存在
            DLinkedNode *node = new DLinkedNode(key, value);
            _cache[key] = node;
            addToHead(node);
            _size++;
            if(_size > _capacity){
                DLinkedNode *removed = removeTail();
                _cache.erase(removed->key);
                // 防止内存泄露
                delete removed;
                --_size;
            }
        }else{
            // key 存在
            DLinkedNode *node = _cache[key];
            node->value = value;
            moveToHead(node);
        }
    }
};
```
