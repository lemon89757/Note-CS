# 回溯算法
> [代码随想录](https://www.programmercarl.com/%E5%9B%9E%E6%BA%AF%E7%AE%97%E6%B3%95%E7%90%86%E8%AE%BA%E5%9F%BA%E7%A1%80.html#%E9%A2%98%E7%9B%AE%E5%88%86%E7%B1%BB)

## 基础
- 回溯是递归的副产品，只要有递归就会有回溯。因此，一般来说回溯函数就是递归函数，它俩指的是同一个东西。
- 回溯法的效率：回溯法并不是高效的算法，因为**回溯的本质是穷举**，穷举所有可能，然后选出想要的答案。**如果想要回溯算法高效一些，可以加一些剪枝操作**，但也改变不了回溯法就是穷举的本质。
- for 循环横向遍历，递归纵向遍历，回溯不断调整结果集
- 回溯法解决的问题：
  - 组合问题：N 个数里面按一定规则找出 K 个数的集合
  - 切割问题：一个字符串按一定规则有几种切割方式
  - 子集问题：一个 N 个数的集合里有多少符合条件的子集
  - 排列问题：N 个数按一定规则全排列，有几种排列方式
  - 棋盘问题：N 皇后，解数独等等

1. 通过递归代替需要个 k 个 for 循环的写法，穷举 k 个数的组合。为避免出现重复，通过 startIndex 记录开始位置。例如 startIndex = 1 时，其他元素可以为 2...n; 当 startIndex = 2 时，其他元素可以为 3...n。优化剪枝时，需要考虑的是在当前所记录的组合中已有多少个元素，以及剩余元素与当前组合的元素个数是否还能满足数量 k 的要求。
```C++
class Solution{
private:
    vector<int> path;
    vector<vector<int>> res;

    void backtracking(int n, int k, int startIndex){
        if(path.size() == k){
            res.emplace_back(path);
            return;
        }

        // 剪枝写法
        // for(int i = startIndex; i < n - (k - path.size()) + 1; i++)
        for(int i = startIndex; i < n; i++){
            path.push_back(i);
            backtracking(n, k, i + 1);
            path.pop_back();
        }
    }
public:
    vector<vector<int>> combine(int n, int k){
        res.clear();
        path.clear();
        backtracking(n, k, 1);
        return res;
    }
}
``` 
2. 提供的数组中的元素允许重复选取，且满足元素值是大于等于 1 的（当 `target` 为 0，如果元素值不满足上述条件，在元素允许重复选取的情况下，可能导致组合数量为无穷个的情况）。即使元素允许重复选取，但目标的还是求**组合**的数量，因此还是需要避免组合重复的情况（去重）。**对于组合问题，如果是同一个集合来求组合的话，就需要 `startIndex`**。
```C++
/* 无剪枝优化版本 */
class Solution{
private:
    vector<int> group;
    vector<vector<int>> res;

    void backtracking(const vector<int> &candidates, int target, int startIndex){
        if(target < 0) return;
        if(target == 0){
            res.emplace_back(group);
            return;
        }

        for(int i = startIndex; i < candidates.size(); i++){
            group.push_back(candidates[i]);
            backtracking(candidates, target - candidates[i], i);
            group.pop_back();
        }
    }

public:
    vector<vector<int>> combinationSum(vector<int> &candidates, int target){
        group.clear();
        res.clear();
        backtracking();
        return res;
    }
}
```
**在求和问题中，排序后剪枝是常见的套路**。
```C++
class Solution{
private:
    vector<int> group;
    vector<vector<int>> res;

    void backtracking(const vector<int> &candidates, int target, int startIndex){
        if(target == 0){
            res.emplace_back(group);
            return;
        }

        // 进行剪枝，当 target 不满足要求时不进入下一次递归
        for(int i = startIndex; i < candidates.size() && target - candidates[i] >= 0; i++){
            group.push_back(candidates[i]);
            backtracking(candidates, target - candidates[i], i);
            group.pop_back();
        }
    }

public:
    vector<vector<int>> combinationSum(vector<int> &candidates, int target){
        group.clear();
        res.clear();
        // 先排序后进行组合
        sort(candidates.begin(), candidates.end());
        backtracking(candidates, target, 0);
        return res;
    }
}
```
3. 数组中可能包含重复元素，且每个元素在组合中只能使用一次。由于包含重复元素，因此需要考虑除上述  `startIndex` 以外的其他去重手段。另外，初始数组是无序的，为了去重需要给数组先进行排序。从而，去重的逻辑是：深层递归时不需要去重，横向遍历时需要去重，去重的方式是：当前元素与前一元素相等，且已经处理过前一元素时需要跳过当前元素。即**同一树层上使用过的需要去重，同一树枝上的都是一个组合里的元素，不用去重**。
```C++
class Solution{
private:
    vector<int> group;
    vector<int> res;

    /* 方法 1: 使用 used 数组进行标记 */
    void backtracking(const vector<int> &candidates, int target, int startIndex, vector<bool> &used){
        if(target == 0){
            res.emplace_back(group);
            return;
        }

        for(int i = startIndex; i < candidates.size() && target - candidates[i] >= 0; i++){
            if(i > 0 && candidates[i] == candidates[i - 1] && !used[i - 1]) continue;

            group.push_back(candidates[i]);
            used[i] = true;
            backtracking(candidates, target - candidates[i], i + 1, used);
            used[i] = false;
            group.pop_back();
        }

    }
    /* 方法 2：结合 startIndex 进行去重 */
    void backtracking(const vector<int> &candidates, int target, int startIndex){
        if(target == 0){
            res.emplace_back(group);
            return;
        }

        for(int i = startIndex; i < candidates.size() && target - candidates[i] >= 0; i++){
            // 去重逻辑
            if(i > startIndex && i > 0 && candidates[i - 1] == candidates[i]) continue;
            group.push_back(candidates[i]);
            backtracking(candidates, target - candidates[i], i + 1);
            group.pop_back();
        }
    }

public:
    vector<vector<int>> combinationSum2(vector<int> &candidates, int target){
        group.clear();
        res.clear();
        // 先排序
        sort(candidates.begin(), candidates.end());
        // 方法 1
        vector<bool> used(candidates.size(), false);
        backtracking(candidates, target, 0, used);
        // 方法 2
        backtracking(candidates, target, 0);

        return res;
    }
}
```
4. 分割会文串是*切割*问题的一种，而**切割问题也可以作为一种特殊的组合问题来处理，特殊之处在于切割不能选取非连续元素**。因此，本题同样可以采用解组合问题的方式来处理：通过递归来选择切割点，*切割点*通过 `startIndex` 来标记，每一种分割方式所得到的回文串通过 `vector<string>` 来记录，因此最终的结果形式为 `vector<vector<string>>`；递归的终止条件为 `startIndex` 超过了需要分割的对象，即 `startIndex >= s.size()`。**模拟切割线，startIndex 是上一层已经确定了的分割线，i 是这一层试图寻找的新分割线**。
```C++
class Solution{
private:
    vector<string> path;
    vector<vector<string>> res;
    // isPalindrome 的第一个索引为子串的起始位置，第二个索引为子串的结束位置
    vector<vector<bool>> isPalindrome;

    /* 通过动态规划方式事先一次性计算出 s 中的任意位置的子串是否为回文串，而动态规划也避免了重复的计算。因为回文串的条件可以总结为 s[start]=start[end] 且 s[start + 1 : end - 1] 为回文串。 */
    bool computePalindrome(const string &s){
        isPalindrome.resize(s.size(), vector<bool>(s.size(), false));
        for(int i = s.size() - 1; i >= 0; i--){
            // 需要倒序计算，保证在 i 进行时，i + 1 已经计算好了
            // 在后序使用 isPalindrome 时是以：
            // if(isPalindrome[startIndex][i]) 的形式，即以 startIndex 为起始位置，以 i 为 结束位置的字串是否为回文串。
            // todo：应该也可以顺序计算 i 的方式来设计函数
            for(int j = i; j < s.size(); j++){
                if(j == i){
                    isPalindrome[i][j] = true;
                }else if(j - i = 1){
                    isPalindrome[i][j] = s[i] == s[j];
                }else{
                    // 前一个字串是 i + 1 到 j - 1 所在区间
                    isPalindrome[i][j] = (s[i] == s[j] && isPalindrome[i + 1][j - 1]);
                }
            }
        }
    }

    /* 通过顺序计算 i 的方式来设计函数 
     * 此时 j 是子串的起始位置，i 是子串的结束位置
     */
    void computePalindrome(const string &s){
        isPalindrome.resize(s.size(), vector<bool>(s.size(), false));
        for(int i = 0; i < s.size(); i++){
            for(int j = i; j >= 0; j--){
                if(j == i){
                    isPalindrome[j][i] = true;
                }else if(i - j == 1){
                    isPalindrome[j][i] = s[i] == s[j];
                }else{
                    // 前一个字串是 j + 1 到 i - 1 所在的区间
                    isPalindrome[j][i] = (s[i] == s[j] && isPalindrome[j + 1][i - 1]);
                }
            }
        }
    }

    /* 通过双指针进行回文串的判断 */
    bool isPalindrome(const string &s, int start, int end){
        for(int i = start, j = end; i < j; i++, j--){
            if(s[i] != s[j]) return false;
        }
        return true;
    }

    void backtracking(const string &s, int startIndex){
        // 如果起始位置已经大于等于 s 的大小，说明已经找到了组分割方案了。
        if(startIndex >= s.size()){
            res.emplace_back(path);
            return;
        }

        for(int i = startIndex; i < s.size(); i++){
            // 模拟切割线，startIndex 是上一层已经确定了的分割线，i 是这一层试图寻找的新分割线
            if(isPalindrome(s, startIndex, i)){
                path.push_back(s.substr(startIndex, i - startIndex + 1));
            }else{
                // 不是回文，跳过。不再向深层递归。
                continue;
            }

            backtracking(s, i + 1);
            path.pop_back();
        }
    }

public:
    vector<vector<string>> partition(string s){
        path.clear();
        res.clear();
        backtracking(s, 0);
        return res;
    }
}
```
