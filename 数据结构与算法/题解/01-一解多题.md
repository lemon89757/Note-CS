   2.3 560.和为 `k` 的子数组：给定一个整数数组 `nums` 和一个整数 `k`，请统计并返回该数组中和为 `k` 的子数组的个数。其中：$1 <= nums.length <= 2 * 10^4; -1000 <= nums[i] <= 1000; -10^7 <= k <= 10^7$。
   ```C++
   /* 滑动窗口 */
   class Solution{
   public:
      int subarraySum(vector<int> &nums, int k){
         if(nums.size() == 1) return nums[0] == k;
         int left = 0, right;
         int sum = nums[left];
         int res= 0;
         if(sum == k) res++;
         for(right = 1; right < nums.size(); right++){
            sum += nums[right];
            while(sum > k && left < right){
               sum -= nums[left];
               left++;
            }
            if(k == sum) res++;
         }

         // right 移动到末端后，在此时的 [left, right] 区间内可能还存在满足条件的子数组
         // 此时 right = nums.size()，因此取 left < right - 1 要让窗口中至少有一个元素
         while(left < right - 1){
            sum -= nums[left];
            left++;
            if(sum == k) res++;
         }
         return res;
      }
   }
   ```
   因为数组既不是有序的，且其中元素还有正负两种情况，因此当滑动窗口中元素的和与 `k` 的大小关系并不能决定是移动左指针还是移动右指针。eg: 当 `nums = [28,54,7,-70,22,65,-6]，k = 100` 时，滑动窗口不能得到正确的答案 1。当最后一个元素被累加前，滑动窗口的元素和已经超过了 `k` 值，因此会导致左指针移动，从而无法得到正确结果。
   ```C++
   /* 枚举，暴力解法 */
   class Solution{
   public:
      int subarraySum(vector<int> &nums, int k){
         int res = 0;
         for(int i = 0; i < nums.size(); i++){
            int sum = 0;
            for(int j = i; j < nums.size(); j++){
               sum += nums[j];
               if(sum == k) res++;
            }
         }
         return res;
      }
   }
   ```
   时间复杂度为 $O(n^2)$，leetcode 超出时间限制。避免子数组每次都要重新计算和，可以建立递推关系：$sum[i] = sum[i - 1] + nums[i]$。于是，在`[i, j]` 上 `nums` 的和满足条件，即 `nums` 在子区间 `[i, j]` 上的和为 `k` 满足：**$sum[j] - sum[i - 1] = k$**。从而，在遍历 `nums` 的过程中，通过在哈希表中记录累加值 `sum[i]` 的情况，并在遍历到 `i + 1` 时，查看是否有满足 $sum[i + 1] - k$ 的哈希表 `key` 并做相应统计即可。
   ```C++
   /* 前缀和 + 哈希表优化 */
   class Solution{
   public:
      int subarraySum(vector<int> &nums, int k){
         int res = 0, sum = 0;
         unordered_map<int, int> mp;
         // 需要让 sum 本身等于k时也作为子数组而成立
         mp[0] = 1;
         for(int i = 0; i < nums.size(); i++){
            sum += nums[i];
            // 在哈希表中检查某个 key 是否存在使用：
            // mp.find(key) == mp.end();
            // 而 mp.count(key) 则是检查 key 在容器中出现的次数
            // 由于在 unordered_map 中不存在两个相同的 key，因此 count 方法返回 1 or 0
            // if(mp.count(sum - k) != 0){
            //   res += mp[sum - k];
            // }
            if(mp.find(sum - k) != mp.end()){
               res += mp[sum - k];
            }
            mp[sum]++;
         }
         return res;
      }
   }
   ```
